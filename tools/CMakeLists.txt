set(UNLEASHED_RECOMP_IOS OFF)
if (CMAKE_SYSTEM_NAME STREQUAL "iOS" OR IOS)
	set(UNLEASHED_RECOMP_IOS ON)
endif()

if (NOT UNLEASHED_RECOMP_IOS)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/file_to_c)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/bc_diff)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/fshasher)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/x_decompress)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/x_content_extract)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/XenonRecomp)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/XenosRecomp)

	set(UNLEASHED_RECOMP_METAL_SHADER_SDK "macosx" CACHE STRING "Apple SDK used by the host game shader compiler")
	set_property(CACHE UNLEASHED_RECOMP_METAL_SHADER_SDK PROPERTY STRINGS macosx iphoneos)
	if (APPLE AND UNLEASHED_RECOMP_METAL_SHADER_SDK STREQUAL "iphoneos")
		# Compile a build-local copy so the pinned tool submodule stays unchanged.
		set(AIR_SOURCE_DIR "${UNLEASHED_RECOMP_TOOLS_ROOT}/XenosRecomp/XenosRecomp")
		file(READ "${AIR_SOURCE_DIR}/air_compiler.cpp" AIR_SOURCE)
		string(REPLACE "\"macosx\"" "\"iphoneos\"" AIR_SOURCE "${AIR_SOURCE}")
		string(REPLACE "\"metal\", \"-o\"" "\"metal\", \"-mios-version-min=15.0\", \"-fmodules-cache-path=${CMAKE_BINARY_DIR}/metal-module-cache\", \"-o\"" AIR_SOURCE "${AIR_SOURCE}")
		string(REPLACE "char sourcePathTemplate[PATH_MAX] = \"/tmp/xenos_metal_XXXXXX.metal\";"
			"char sourcePathTemplate[PATH_MAX];\n    const char* temporaryDirectory = getenv(\"TMPDIR\");\n    snprintf(sourcePathTemplate, sizeof(sourcePathTemplate), \"%s/xenos_metal_XXXXXX.metal\", temporaryDirectory ? temporaryDirectory : \"/tmp\");"
			AIR_SOURCE "${AIR_SOURCE}")
		file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/air_compiler_ios.cpp" "${AIR_SOURCE}")
		get_target_property(XENOS_SOURCES XenosRecomp SOURCES)
		list(REMOVE_ITEM XENOS_SOURCES air_compiler.cpp)
		set_property(TARGET XenosRecomp PROPERTY SOURCES "${XENOS_SOURCES}")
		target_sources(XenosRecomp PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/air_compiler_ios.cpp")
		target_include_directories(XenosRecomp PRIVATE "${AIR_SOURCE_DIR}")
	elseif (NOT UNLEASHED_RECOMP_METAL_SHADER_SDK STREQUAL "macosx")
		message(FATAL_ERROR "Unsupported Metal shader SDK: ${UNLEASHED_RECOMP_METAL_SHADER_SDK}")
	endif()
else()
	set(THIRDPARTY_ROOT ${UNLEASHED_RECOMP_TOOLS_ROOT}/XenonRecomp/thirdparty)
	add_subdirectory(${THIRDPARTY_ROOT}/disasm)
	add_subdirectory(${UNLEASHED_RECOMP_TOOLS_ROOT}/XenonRecomp/XenonUtils)
endif()
